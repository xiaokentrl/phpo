  return list.map(m => {

    if(m.global){
      host = state.env[m.key] || '';
    } else if(m.sub){
      host = `${root}/${version}/${m.sub}`;
      if(m.from) host += `/${m.from}`;
    }
    return { ...m, host };
  });

function defaultVhost(domain, port, php, hostRoot, rewriteKey, customRule){
  const rw = REWRITE_PRESETS[rewriteKey] || REWRITE_PRESETS.none;
  const rule = (typeof customRule === 'string' && customRule.trim()) ? customRule : rw.rule;
  const phpUpstream = `php-${php}-fpm`;
  const ruleLines = rule.split('\n').map(l => '    ' + l).join('\n');
  const containerRoot = hostToContainer(hostRoot);
  return `server {
    listen ${port};
    server_name ${domain};
    root ${containerRoot};
    index index.php index.html;
${ruleLines}
    location ~ \\.php$ {
        resolver 127.0.0.11 valid=10s ipv6=off;
        set $php_upstream ${phpUpstream}:9000;
        fastcgi_pass $php_upstream;
        fastcgi_index index.php;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
    }
}`;
}

function parseVhost(content){
  const result = { port: null, root: null };
  if(!content) return result;
  const text = String(content);
  const listenMatch = text.match(/^\s*listen\s+(\d+)/m);
  if(listenMatch){
    const p = parseInt(listenMatch[1], 10);
    if(!isNaN(p) && p >= 1 && p <= 65535) result.port = p;
  }
  const rootMatch = text.match(/^\s*root\s+([^;]+);/m);
  if(rootMatch) result.root = rootMatch[1].trim();
  return result;
}
function replaceListen(content, port){
  if(!content) return content;
  return String(content).replace(/^(\s*listen\s+)\d+(\s[^;]*)?;/m, `$1${port}$2;`);
}
/* ★ FIX（总纲 #1）：上游名 php-{version}-fpm:9000 */
function replacePhpUpstream(content, php){
  if(!content) return content;
  return String(content).replace(/set\s+\$php_upstream\s+[^;]+;/, () => `set $php_upstream php-${php}-fpm:9000;`);
}

const VHosts = {
  compute(site){
    if(!site) return '';
    return defaultVhost(site.domain, site.port, site.php, site.root, site.rewrite, site.rewriteRule);
  },
  get(domain){
    const site = state.sites.find(s => s.domain === domain);
    if(!site) return '';
    const cached = state.vhosts[domain];
    if(typeof cached === 'string' && cached.length) return cached;
    return this.compute(site);
  },
  regenerate(domain, { persist = true } = {}){
    const site = state.sites.find(s => s.domain === domain);
    if(!site){ delete state.vhosts[domain]; if(persist) persistState(); return ''; }
    state.vhosts[domain] = this.compute(site);
    delete site.vhostCustomized;
    if(persist) persistState();
    return state.vhosts[domain];
  },
  setContent(domain, content, { persist = true } = {}){
    const site = state.sites.find(s => s.domain === domain);
    if(!site) return '';
    state.vhosts[domain] = String(content ?? '');
    site.vhostCustomized = true;
    const parsed = parseVhost(content);
    if(parsed.port != null) site.port = parsed.port;
    if(parsed.root) site.root = containerToHost(parsed.root);
    if(persist) persistState();
    return state.vhosts[domain];
  },
  applyPort(domain, port, { persist = true } = {}){
    const site = state.sites.find(s => s.domain === domain);
    if(!site) return '';
    site.port = port;
    const cached = state.vhosts[domain];
    if(site.vhostCustomized && typeof cached === 'string' && cached.length){
      state.vhosts[domain] = replaceListen(cached, port);
    } else {
      state.vhosts[domain] = this.compute(site);
      delete site.vhostCustomized;
    }
    if(persist) persistState();
    return state.vhosts[domain];
  },
  applyPhp(domain, php, { persist = true } = {}){
    const site = state.sites.find(s => s.domain === domain);
    if(!site) return '';
    site.php = php;
    const cached = state.vhosts[domain];
    if(site.vhostCustomized && typeof cached === 'string' && cached.length){
      state.vhosts[domain] = replacePhpUpstream(cached, php);
    } else {
      state.vhosts[domain] = this.compute(site);
      delete site.vhostCustomized;
    }
    if(persist) persistState();
    return state.vhosts[domain];
  },
  applyRewrite(domain, preset, rule, { persist = true } = {}){
    const site = state.sites.find(s => s.domain === domain);
    if(!site) return '';
    site.rewrite = preset;
    site.rewriteRule = rule || '';
    state.vhosts[domain] = this.compute(site);
    delete site.vhostCustomized;
    if(persist) persistState();
    return state.vhosts[domain];
  },
  remove(domain, { persist = true } = {}){
    delete state.vhosts[domain];
    if(persist) persistState();
  },
  rebuildAll({ persist = true } = {}){
    const domains = new Set(state.sites.map(s => s.domain));
    for(const d of Object.keys(state.vhosts)){
      if(!domains.has(d)) delete state.vhosts[d];
    }
    for(const site of state.sites){
      const cached = state.vhosts[site.domain];
      if(!site.vhostCustomized || typeof cached !== 'string' || !cached.length){
        state.vhosts[site.domain] = this.compute(site);
        delete site.vhostCustomized;
      }
    }
    if(persist) persistState();
  },
};

function normPath(p){ return String(p||'').trim().replace(/\/+/g,'/').replace(/\/+$/,''); }
function hasTraversal(p){ return /(^|\/)\.\.(\/|$)/.test(String(p||'')); }

/* ★ FIX #5：占用端口集合包含站点端口，可排除指定域名 */
function collectUsedPorts(excludeDomains = []){
  const used = new Map();
  for(const kind of ['mysql','pgsql','redis']){
    for(const v of (state.installed[kind]||[])){
      const key = `${kind.toUpperCase()}_${v.replace(/\./g,'')}_PORT`;
      const p = parseInt(state.env[key],10);
      if(!isNaN(p) && !used.has(p)) used.set(p, `${kind} ${v}`);
    }
  }
  for(const s of state.sites){
    if(excludeDomains.includes(s.domain)) continue;
    const p = parseInt(s.port,10);
    if(!isNaN(p) && !used.has(p)) used.set(p, `site ${s.domain}`);
  }
  return used;
}

/* ★ FIX #5：从 desired+1 向上到 65535，再回绕 1..desired-1，顺延寻找首个可用端口（无窗口上限） */
function findNextAvailablePort(desired, maxPort = 65535, minPort = 1, used){
  for(let p = desired + 1; p <= maxPort; p++){
    if(!used.has(p)) return p;
  }
  for(let p = minPort; p < desired; p++){
    if(!used.has(p)) return p;
  }
  return null;
}

/* ★ FIX #3：版本号仅做路径安全校验（禁止 / \ 空白 控制符 ..） */
function validateVersion(kind, version){
  const v = String(version||'').trim();
  if(!v) return { ok:false, msg:PF.versionInvalid };
  if(v === '.' || v === '..' || v.includes('..')) return { ok:false, msg:`${PF.versionInvalid}: ${v}` };
  if(/[/\\\x00-\x1f\s]/.test(v)) return { ok:false, msg:`${PF.versionInvalid}: ${v}` };
  if(v.length > 128) return { ok:false, msg:`${PF.versionInvalid}: ${v}` };
  return { ok:true, value:v };
}

/* ★ FIX #5：validatePort 支持 autoAdvance 顺延链（站点端口占用顺延首个可用，不报错） */
function validatePort(port, opts = {}){
  const s = String(port ?? '').trim();
  if(!/^\d{1,5}$/.test(s)) return { ok:false, msg:PF.portInvalid };
  const n = parseInt(s, 10);
  if(n < 1 || n > 65535) return { ok:false, msg:PF.portInvalid };
  const exclude = (opts.exclude == null ? [] :
    (Array.isArray(opts.exclude) ? opts.exclude : [opts.exclude])).map(x => parseInt(x,10));
  if(exclude.includes(n)) return { ok:true, value:n };
  const used = collectUsedPorts(opts.excludeDomains || []);
  if(!used.has(n)) return { ok:true, value:n };
  if(opts.autoAdvance){
    const next = findNextAvailablePort(n, 65535, 1, used);
    if(next != null) return { ok:true, value:next, adjusted:true, original:n };
  }
  return { ok:false, msg:`${PF.portInUse}: ${n} (${used.get(n)})` };
}
function validateDomain(domain){
  const d = String(domain||'').trim().toLowerCase();
  if(!d || d.length > 253) return { ok:false, msg:PF.domainInvalid };
  if(!/^[a-z0-9]([a-z0-9-]*[a-z0-9])?(\.[a-z0-9]([a-z0-9-]*[a-z0-9])?)*$/.test(d))
    return { ok:false, msg:`${PF.domainInvalid}: ${d}` };
  return { ok:true, value:d };
}
/* ★ FIX #4：站点根允许在 WWW_ROOT 之外（降级为 warning） */
function validateSiteRoot(root){
  const r = normPath(root);
  if(!r) return { ok:false, msg:PF.rootEmpty };
  if(hasTraversal(r)) return { ok:false, msg:PF.pathTraversal };
  const www = normPath(state.env.WWW_ROOT);
  if(!(r === www || r.startsWith(www + '/'))){
    return { ok:true, value:r, outsideWww:true };
  }
  return { ok:true, value:r };
}
function validateExt(name){
  return /^[a-zA-Z0-9._-]+$/.test(String(name||'').trim());
}

function preflight(action, ctx = {}){
  const errors = [];
  const warnings = [];
  const adjusted = {};
  const c = ctx || {};

  if(state.task && state.task.status === TASK_STATUS.RUNNING) errors.push(PF.taskBusy);

  const NEEDS_HOME = new Set([
    'install','uninstall','service-stop','service-start','update-config',
    'site-add','site-remove','site-port','site-vhost','rewrite',
    'extensions','service-config','backup','restore','offline-prune',
  ]);
  if(NEEDS_HOME.has(action) && !state.dirReady.PHPO_HOME) errors.push(PF.homeNotReady);

  switch(action){

    case 'install': {
      const { kind, version, port, extensions } = c;
      if(!SVC_META[kind]){ errors.push(PF.svcMissing); break; }
      const vv = validateVersion(kind, version);
      if(!vv.ok){ errors.push(vv.msg); break; }
      if((state.installed[kind]||[]).includes(vv.value))
        errors.push(`${PF.versionDup}: ${kind} ${vv.value}`);
      if(kind === 'nginx' && (state.installed.nginx||[]).length)
        errors.push(PF.nginxSingle);
      if(port != null && String(port).trim() !== ''){
        const pp = validatePort(port);
        if(!pp.ok) errors.push(pp.msg);
      }
      /* ★ FIX #2：移除密码长度校验 */
      if(extensions){
        const exts = String(extensions).split(',').map(x=>x.trim()).filter(Boolean);
        for(const e of exts){
          if(!validateExt(e)){ errors.push(`${PF.extInvalid}: ${e}`); break; }
        }
      }
      break;
    }

    case 'uninstall': {
      const { kind, version } = c;
      if(!SVC_META[kind]){ errors.push(PF.svcMissing); break; }
      if(!(state.installed[kind]||[]).includes(version)){
        errors.push(`${PF.notInstalled}: ${kind} ${version}`); break;
      }
      if(kind === 'php'){
        const used = state.sites.filter(s=>s.php === version).map(s=>s.domain);
        if(used.length) errors.push(`${PF.hasDependents}: PHP ${version} ← ${used.join(', ')}`);
        if((state.installed.php||[]).length <= 1) errors.push(PF.lastPhp);
      }
      if(kind === 'nginx' && state.sites.length)
        errors.push(`${PF.hasDependents}: ${state.sites.length} 个站点依赖 Nginx`);
      break;
    }

    case 'service-stop': {
      const { kind, version } = c;
      if(!SVC_META[kind]){ errors.push(PF.svcMissing); break; }
      if(!(state.installed[kind]||[]).includes(version)){
        errors.push(`${PF.notInstalled}: ${kind} ${version}`); break;
      }
      if(!isServiceRunning(kind, version)){
        errors.push(`${PF.notRunning}: ${kind} ${version}`); break;
      }
      if(kind === 'php'){
        const used = state.sites.filter(s=>s.php === version).map(s=>s.domain);
        if(used.length) warnings.push(`以下站点正在使用 PHP ${version}：${used.join(', ')}`);
      }
      if(kind === 'nginx' && state.sites.length)
        warnings.push(`${state.sites.length} 个站点依赖 Nginx，停用后无法访问`);
      if(kind === 'mysql' || kind === 'pgsql' || kind === 'redis')
        warnings.push(`停用 ${kind} ${version} 将中断正在使用该服务的应用`);
      break;
    }

    case 'service-start': {
      const { kind, version } = c;
      if(!SVC_META[kind]){ errors.push(PF.svcMissing); break; }
      if(!(state.installed[kind]||[]).includes(version)){
        errors.push(`${PF.notInstalled}: ${kind} ${version}`); break;
      }
      if(isServiceRunning(kind, version)){
        errors.push(`${PF.isRunning}: ${kind} ${version}`); break;
      }
      break;
    }

    case 'update-config': {
      const { kind, version, field, newValue } = c;
      if(!SVC_META[kind]){ errors.push(PF.svcMissing); break; }
      if(!(state.installed[kind]||[]).includes(version)){
        errors.push(`${PF.notInstalled}: ${kind} ${version}`); break;
      }
      if(field === 'port'){
        let cur;
        if(kind === 'nginx') cur = state.env.NGINX_PORT;
        else cur = state.env[`${kind.toUpperCase()}_${version.replace(/\./g,'')}_PORT`];
        const pp = validatePort(newValue, { exclude: cur });
        if(!pp.ok) errors.push(pp.msg);
      }
      /* ★ FIX #2：移除密码长度校验 */
      break;
    }

    case 'site-add': {
      const { domain, port, php, root } = c;
      if(!(state.installed.php||[]).length){ errors.push(PF.phpNeeded); break; }
      if(!(state.installed.nginx||[]).length){ errors.push(PF.nginxNeeded); break; }
      const dd = validateDomain(domain);
      if(!dd.ok){ errors.push(dd.msg); break; }
      if(state.sites.some(s=>s.domain === dd.value))
        errors.push(`${PF.domainExists}: ${dd.value}`);
      /* ★ FIX #5：站点端口冲突在 1–65535 内顺延首个可用（不报错） */
      if(port != null){
        const pp = validatePort(port, { autoAdvance: true });
        if(!pp.ok) errors.push(pp.msg);
        else if(pp.adjusted){
          adjusted.port = pp.value;
          warnings.push(PF.portAdvance.replace('{from}', pp.original).replace('{to}', pp.value));
        }
      }
      if(php && !(state.installed.php||[]).includes(php))
        errors.push(`${PF.notInstalled}: PHP ${php}`);
      if(root){
        const rr = validateSiteRoot(root);
        if(!rr.ok) errors.push(rr.msg);
        else {
          /* ★ FIX #4：WWW_ROOT 之外为 warning 而非 error */
          if(rr.outsideWww) warnings.push(`${PF.rootOutsideWww}: ${rr.value}`);
          if(state.sites.some(s=>normPath(s.root) === rr.value))
            errors.push(`${PF.rootDuplicated}: ${rr.value}`);
        }
      }
      break;
    }

    case 'site-remove': {
      if(!state.sites.some(s=>s.domain === c.domain)) errors.push(`${PF.siteMissing}: ${c.domain}`);
      break;
    }

    case 'site-port': {
      const site = state.sites.find(s=>s.domain === c.domain);
      if(!site){ errors.push(`${PF.siteMissing}: ${c.domain}`); break; }
      /* ★ FIX #5：站点端口编辑同样顺延首个可用（不报错） */
      const pp = validatePort(c.newValue, {
        exclude: site.port,
        excludeDomains: [c.domain],
        autoAdvance: true,
      });
      if(!pp.ok) errors.push(pp.msg);
      else if(pp.adjusted){
        adjusted.port = pp.value;
        warnings.push(PF.portAdvance.replace('{from}', pp.original).replace('{to}', pp.value));
      }
      break;
    }

    case 'site-vhost': {
      const { domain, content, php } = c;
      const site = state.sites.find(s=>s.domain === domain);
      if(!site){ errors.push(`${PF.siteMissing}: ${domain}`); break; }
      if(content != null && !String(content).trim()) errors.push(PF.configEmpty);
      if(php && !(state.installed.php||[]).includes(php))
        warnings.push(`${PF.notInstalled}: PHP ${php}（站点配置仍可保存，但需安装该版本才能生效）`);
      break;
    }

    case 'php-switch': {
      const { domain, newPhp } = c;
      const site = state.sites.find(s=>s.domain === domain);
      if(!site){ errors.push(`${PF.siteMissing}: ${domain}`); break; }
      if(!(state.installed.php||[]).includes(newPhp))
        errors.push(`${PF.notInstalled}: PHP ${newPhp}`);
      break;
    }

    case 'rewrite': {
      const site = state.sites.find(s=>s.domain === c.domain);
      if(!site){ errors.push(`${PF.siteMissing}: ${c.domain}`); break; }
      if(!(state.installed.nginx||[]).length) errors.push(PF.nginxNeeded);
      break;
    }

    case 'extensions': {
      const { version, finalExts } = c;
      if(!(state.installed.php||[]).includes(version)){
        errors.push(`${PF.notInstalled}: PHP ${version}`); break;
      }
      for(const e of (finalExts||[])){
        if(!validateExt(e)){ errors.push(`${PF.extInvalid}: ${e}`); break; }
      }
      break;
    }

    case 'service-config': {
      const { kind, version, files } = c;
      if(!SVC_META[kind]){ errors.push(PF.svcMissing); break; }
      if(!(state.installed[kind]||[]).includes(version)){
        errors.push(`${PF.notInstalled}: ${kind} ${version}`); break;
      }
      if(!files || !files.length) errors.push(PF.configEmpty);
      break;
    }

    case 'backup': {
      if(!(state.installed.php||[]).length && !state.sites.length && !(state.installed.nginx||[]).length)
        warnings.push('当前环境为空，备份将生成空归档');
      break;
    }

    case 'restore': {
      if(!state.backups.some(b=>b.file === c.file))
        errors.push(`${PF.backupMissing}: ${c.file}`);
      break;
    }

    case 'backup-delete': {
      if(!state.backups.some(b=>b.file === c.file))
        errors.push(`${PF.backupMissing}: ${c.file}`);
      break;
    }

    case 'offline-prune': {
      if(!state.offline.trees.some(x=>x.svc === c.svc && x.ver === c.ver))
        errors.push(`${PF.offlineMissing}: ${c.svc}/${c.ver}`);
      break;
    }
  }

  return { ok: errors.length === 0, errors, warnings, adjusted };
}

function runGuardedTask(action, ctx, args, label, meta, opts = {}){
  const check = preflight(action, ctx);
  if(!check.ok){
    toast(check.errors.join('\n'), 'err', 4600);
    return false;
  }
  if(!opts.skipWarnings && check.warnings.length){
    openDangerConfirm({
      title: label || String(action),
      warnings: check.warnings.map(w => ({ text: w })),
      confirmLabel: t('common.confirm'),
      onConfirm: () => runTask(args, label, meta),
    });
    return true;
  }
  runTask(args, label, meta);
  return true;
}

  function renderAll(){
    root.innerHTML = fullHtml();
    setupModalResize($('.modal', root));
    bindEvents();
    if(step === 1){ const el = $('#wiz-home', root); if(el){ el.focus(); el.setSelectionRange(el.value.length, el.value.length); } }
    else if(step === 2){ const el = $('#wiz-www', root); if(el){ el.focus(); el.setSelectionRange(el.value.length, el.value.length); } }
  }
  function updateHomeTree(){ const box = $('#wiz-home-tree', root); if(!box) return; const h = (homeVal.v.trim() || DEFAULT_HOME).replace(/\/+$/,''); box.innerHTML = homeTreeHtml(h); }
  function updateWwwTree(){ const box = $('#wiz-www-tree', root); if(!box) return; const w = (wwwVal.v.trim() || DEFAULT_WWW).replace(/\/+$/,''); box.innerHTML = wwwTreeHtml(w); }

  function onNext(){
    if(step === 1){ const h = homeVal.v.trim(); if(!h){ toast(t('wiz.needHome'), 'err'); $('#wiz-home', root)?.focus(); return; } step = 2; renderAll(); return; }
    if(step === 2){
      const w = wwwVal.v.trim();
      if(!w){ toast(t('wiz.needWww'), 'err'); $('#wiz-www', root)?.focus(); return; }
      const h = homeVal.v.trim().replace(/\/+$/,'');
      const w2 = w.replace(/\/+$/,'');
      if(h === w2){ toast(t('wiz.samePath'), 'err'); $('#wiz-www', root)?.focus(); return; }
      step = 3; verified = false; renderAll(); return;
    }
  }
  function onPrev(){ if(step > 1){ step--; verified = false; renderAll(); } }

  async function doVerify(){
    if(verifying) return;
    const h = homeVal.v.trim().replace(/\/+$/,'');
    const w = wwwVal.v.trim().replace(/\/+$/,'');
    if(!h){ toast(t('wiz.needHome'), 'err'); return; }
    if(!w){ toast(t('wiz.needWww'), 'err'); return; }
    if(h === w){ toast(t('wiz.samePath'), 'err'); return; }
    verifying = true;
    const verifyBtn = $('#wiz-verify', root);
    if(verifyBtn) verifyBtn.disabled = true;
    const logWrap = $('#wiz-log-wrap', root);
    const logBox = $('#wiz-log', root);
    if(!logBox) return;
    logWrap.hidden = false;
    logBox.innerHTML = `<span class="prompt">$ </span>phpo home ensure ${esc(h)} --www ${esc(w)}\n`;
    const wait = ms => new Promise(r => setTimeout(r, ms));
    const push = (el) => { logBox.appendChild(el); logBox.scrollTop = logBox.scrollHeight; };
    const mkStep = (text) => { const d = document.createElement('div'); d.className = 'step'; d.textContent = '· ' + text; push(d); return d; };
    const mkOk = (text) => { const d = document.createElement('div'); d.className = 'ok'; d.textContent = text; push(d); return d; };

    const l1 = mkStep(`${t('wiz.step1')}：${h}`); await wait(260); l1.className = 'ok'; l1.textContent = `${t('wiz.s3.homeReady')}：${h}`;
    const l2 = mkStep(`${t('wiz.step2')}：${w}`); await wait(260); l2.className = 'ok'; l2.textContent = `${t('wiz.s3.wwwReady')}：${w}  → /var/www`;

    const l3 = mkStep(t('wiz.s3.subStart', {n: HOME_SUBDIRS.filter(s=>s.depth===0).length}));
    await wait(240);
    l3.className = 'ok';
    l3.textContent = '✓ ' + t('wiz.s3.subStart', {n: HOME_SUBDIRS.filter(s=>s.depth===0).length});
    for(const sd of HOME_SUBDIRS){
      mkOk(`    ✓ ${h}/${sd.path}/   ${sd.label}`);
      await wait(55 + Math.random()*65);
    }

    const l4 = mkStep(`${t('wiz.s3.wwwCreate')}：${w}`); await wait(220); l4.className = 'ok'; l4.textContent = `✓ ${t('wiz.s3.wwwCreate')}：${w}`;
    const l5 = mkStep(t('wiz.s3.chmod')); await wait(280); l5.className = 'ok'; l5.textContent = '✓ ' + t('wiz.s3.chmod') + '   u+rwX,go+rX';
    const l6 = mkStep(t('wiz.s3.rw')); await wait(280); l6.className = 'ok'; l6.textContent = '✓ ' + t('wiz.s3.rw') + '   write / read / delete OK';
    await wait(180);
    verifying = false;
    verified = true;
    const vb = $('#wiz-verify', root);
    if(vb){
      const cb = document.createElement('button');
      cb.className = 'btn btn-primary'; cb.id = 'wiz-confirm'; cb.textContent = t('dir.confirm');
      cb.addEventListener('click', doConfirm);
      vb.replaceWith(cb);
    }
  }
  function doConfirm(){
    const h = homeVal.v.trim().replace(/\/+$/,'');
    const w = wwwVal.v.trim().replace(/\/+$/,'');
    Object.assign(state.env, derivePaths(h, w));
    state.dirReady.PHPO_HOME = true;
    persistDirReady();
    VHosts.rebuildAll({ persist: false });
    persistState();
    closeModal();
    onReady && onReady();
  }
  function bindEvents(){
    const nextBtn = $('#wiz-next', root); if(nextBtn) nextBtn.addEventListener('click', onNext);
    const prevBtn = $('#wiz-prev', root); if(prevBtn) prevBtn.addEventListener('click', onPrev);
    const verifyBtn = $('#wiz-verify', root); if(verifyBtn) verifyBtn.addEventListener('click', doVerify);
    const confirmBtn = $('#wiz-confirm', root); if(confirmBtn) confirmBtn.addEventListener('click', doConfirm);
    const homeInput = $('#wiz-home', root);
    if(homeInput){ homeInput.addEventListener('input', ()=>{ homeVal.v = homeInput.value; updateHomeTree(); }); homeInput.addEventListener('keydown', e=>{ if(e.key === 'Enter'){ e.preventDefault(); onNext(); } }); }
    const homeBrowse = $('#wiz-home-browse', root);
    if(homeBrowse){ homeBrowse.addEventListener('click', async ()=>{ await pickDirectory(p => { homeVal.v = p; homeInput.value = p; updateHomeTree(); }); }); }
    const wwwInput = $('#wiz-www', root);
    if(wwwInput){ wwwInput.addEventListener('input', ()=>{ wwwVal.v = wwwInput.value; updateWwwTree(); }); wwwInput.addEventListener('keydown', e=>{ if(e.key === 'Enter'){ e.preventDefault(); onNext(); } }); }
    const wwwBrowse = $('#wiz-www-browse', root);
    if(wwwBrowse){ wwwBrowse.addEventListener('click', async ()=>{ await pickDirectory(p => { wwwVal.v = p; wwwInput.value = p; updateWwwTree(); }); }); }
    $$('[data-modal-close]', root).forEach(b=>b.addEventListener('click', closeModal));
  }
  root = openModalShell('');
  renderAll();
}

function buildScript(args, meta){
  const lines = [{t:'cmd', s:`$ phpo ${args.join(' ')}`}];
  const kind = meta?.kind, version = meta?.version;
  if(meta?.type === 'install'){
    if(MOUNTS[kind]){
      lines.push({t:'meta', s:'[1/6] Precheck'});
      lines.push({t:'ok',   s:'  ✓ docker ready'});
      lines.push({t:'ok',   s:`  ✓ PHPO_HOME=${state.env.PHPO_HOME}`});
      lines.push({t:'meta', s:'[2/6] Create version dirs'});
      for(const sub of (VERSION_SUBDIRS[kind]||[])){
        lines.push({t:'ok', s:`  ✓ ${verRoot(kind, version)}/${sub}/`});
      }
      lines.push({t:'meta', s:'[3/6] Write .env'});
      if(meta.port){
        const key = kind === 'nginx' ? 'NGINX_PORT' : `${kind.toUpperCase()}_${version.replace(/\./g,'')}_PORT`;
        lines.push({t:'ok',   s:`  ✓ ${key}=${meta.port}`});
      }
      if(meta.password){
        const key = `${kind.toUpperCase()}_${version.replace(/\./g,'')}_ROOT_PASSWORD`;
        lines.push({t:'ok',   s:`  ✓ ${key}=****${meta.password.slice(-4)}`});
      }
      if(kind === 'php' && meta.extensions) lines.push({t:'ok', s:`  ✓ EXTENSIONS=${meta.extensions}`});
      lines.push({t:'meta', s:'[4/6] Write config'});
      for(const f of getDefaultFiles(kind, version)){
        lines.push({t:'ok', s:`  ✓ ${state.env.PHPO_HOME}/${f.path}`});
      }
      lines.push({t:'meta', s:'[5/6] Mounts'});
      for(const m of resolveMounts(kind, version)){
        lines.push({t:'ok', s:`  ✓ [${m.mode}] ${m.host} → ${m.to}   (${m.label})`});
      }
      lines.push({t:'meta', s:'[6/6] Start'});
      lines.push({t:'ok',   s:`  ✓ ${kind}-${version} started${meta.port?' on :'+meta.port:''}`});
      lines.push({t:'ok',   s:`✓ ${kind} ${version} installed`});
    } else {
      lines.push({t:'meta', s:'[1/4] Precheck'});
      lines.push({t:'ok',   s:'  ✓ docker ready'});
      lines.push({t:'meta', s:'[2/4] Configure'});

function syncDrawerControls(){
  const task = state.task;
  const running = task?.status === TASK_STATUS.RUNNING;
  const cancelBtn = $('#drawer-cancel');
  if(cancelBtn) cancelBtn.hidden = !running;
  const dot = $('#drawer-dot');
  if(dot){
    let cls = 'drawer-dot';
    if(running) cls += ' running';
    else if(task?.status === TASK_STATUS.SUCCESS) cls += ' success';
    else if(task?.status === TASK_STATUS.FAILED) cls += ' failed';
    dot.className = cls;
  }
  const statusEl = $('#drawer-status');
  if(statusEl){
    if(running) statusEl.textContent = t('task.running');
    else if(task?.status === TASK_STATUS.SUCCESS){
      const dur = ((Date.now() - task.startedAt) / 1000).toFixed(1);
      statusEl.textContent = t('task.complete', {dur});
    }
    else if(task?.status === TASK_STATUS.CANCELLED) statusEl.textContent = t('task.cancelled');
    else if(task?.status === TASK_STATUS.FAILED) statusEl.textContent = t('task.failed');
    else statusEl.textContent = '';
  }
}

function runTask(args, label, meta){
  if(state.task && state.task.status === TASK_STATUS.RUNNING){
    toast(PF.taskBusy, 'err', 2200);
    return;
  }
  if(state.task && state.task.timer){ clearTimeout(state.task.timer); }
  state.task = {
    args,
    label: label || args.join(' '),
    lines: buildScript(args, meta),
    status: TASK_STATUS.RUNNING,
    cursor: 0,
    startedAt: Date.now(),
    meta,
    cancelled: false,
    timer: null,
    abortReason: null,
  };
  const drawer = $('#drawer');
  $('#drawer-label').textContent = state.task.label;
  $('#drawer-cmd').textContent = 'phpo ' + args.join(' ');
  $('#drawer-log').innerHTML = '';
  drawer.classList.add('expanded');
  syncDrawerControls();
  updateTray();
  playTask(meta);
}

function playTask(meta){
  const task = state.task, log = $('#drawer-log');
  function next(){
    if(!state.task || state.task !== task || task.cancelled){
      task.status = TASK_STATUS.CANCELLED;
      task.timer = null;
      syncDrawerControls();
      updateTray();
      return;
    }
    if(task.cursor >= task.lines.length){
      try{
        applyStateChange(meta);
        persistState();
        task.status = TASK_STATUS.SUCCESS;
      }catch(err){
        task.status = TASK_STATUS.FAILED;
        task.abortReason = String((err && err.message) || err);
        console.error('[phpo task failed]', task.abortReason);
      }
      task.timer = null;
      syncDrawerControls();
      updateTray();
      return;
    }
    const entry = task.lines[task.cursor++];
    const div = document.createElement('div');
    let cls = 'log-line';
    if(entry.t==='err') cls+=' err';
    if(entry.t==='ok') cls+=' ok';
    if(entry.t==='cmd') cls+=' cmd';
    if(entry.t==='meta') cls+=' meta';
    if(entry.t==='dim') cls+=' dim';
    div.className = cls;
    div.textContent = entry.s || ' ';
    log.appendChild(div);
    log.scrollTop = log.scrollHeight;
    task.timer = setTimeout(next, entry.t==='cmd'?120:90+Math.random()*200);
  }
  next();
}

function cancelTask(){
  const task = state.task;
  if(!task || task.status !== TASK_STATUS.RUNNING) return;
  task.cancelled = true;
  if(task.timer){ clearTimeout(task.timer); task.timer = null; }
  task.status = TASK_STATUS.CANCELLED;
  task.abortReason = 'user';
  syncDrawerControls();
  updateTray();
  toast(t('task.cancelHint'), 'info', 2600);
}

function applyStateChange(meta){
  if(!meta) return;
  if(meta.type==='install'){
    if(meta.kind==='nginx') state.installed.nginx=[meta.version];
    else if(!state.installed[meta.kind].includes(meta.version)){
      state.installed[meta.kind].push(meta.version);
      state.installed[meta.kind].sort(cmpVer);
      if(meta.kind==='php') state.phpExtensions[meta.version]=['gd','redis','pdo_mysql','mysqli','pgsql','pdo_pgsql','zip','bcmath','intl','opcache','exif','soap','sockets','imagick','xdebug'];
    }
    if(meta.port){
      if(meta.kind==='nginx') state.env.NGINX_PORT = String(meta.port);
      else if(meta.version){ const key = `${meta.kind.toUpperCase()}_${meta.version.replace(/\./g,'')}_PORT`; state.env[key] = String(meta.port); }
    }
    /* ★ FIX #2：允许空密码 —— 只要字段存在就写入 */
    if(meta.password !== undefined && meta.password !== null && meta.version){
      const key = `${meta.kind.toUpperCase()}_${meta.version.replace(/\./g,'')}_ROOT_PASSWORD`;
      state.env[key] = String(meta.password);
    }
  } else if(meta.type==='uninstall'){
    state.installed[meta.kind] = state.installed[meta.kind].filter(v=>v!==meta.version);
    state.stopped[meta.kind] = state.stopped[meta.kind].filter(v => v !== meta.version);
    if(meta.kind === 'php') delete state.phpExtensions[meta.version];
  } else if(meta.type==='service-stop'){
    if(!state.stopped[meta.kind].includes(meta.version)) state.stopped[meta.kind].push(meta.version);
  } else if(meta.type==='service-start'){
    state.stopped[meta.kind] = state.stopped[meta.kind].filter(v => v !== meta.version);
  } else if(meta.type==='update-config'){
    if(meta.field === 'port'){
      if(meta.kind === 'nginx') state.env.NGINX_PORT = String(meta.newValue);
      else { const key = `${meta.kind.toUpperCase()}_${meta.version.replace(/\./g,'')}_PORT`; state.env[key] = String(meta.newValue); }
    } else if(meta.field === 'password'){
      const key = `${meta.kind.toUpperCase()}_${meta.version.replace(/\./g,'')}_ROOT_PASSWORD`;
      state.env[key] = String(meta.newValue);
    }
  } else if(meta.type==='site-add'){
    if(!state.sites.find(s=>s.domain===meta.domain)){
      const hostRoot = meta.root || `${state.env.WWW_ROOT}/${meta.domain}`;
      state.sites.push({ domain:meta.domain, port:meta.port, php:meta.php, root: hostRoot, hosts:false, health:'warn', rewrite:meta.rewrite });
      VHosts.regenerate(meta.domain, { persist: false });
    }
  } else if(meta.type==='site-port'){
    VHosts.applyPort(meta.domain, meta.newValue, { persist: false });
  } else if(meta.type==='site-vhost'){
    VHosts.setContent(meta.domain, meta.content, { persist: false });
  } else if(meta.type==='php-switch'){
    VHosts.applyPhp(meta.domain, meta.newPhp, { persist: false });
    toast(t('sites.switched',{domain: meta.domain, version: meta.newPhp}),'ok',2400);
  } else if(meta.type==='rewrite'){
    VHosts.applyRewrite(meta.domain, meta.preset, meta.rule, { persist: false });
  } else if(meta.type==='site-remove'){
    state.sites = state.sites.filter(s=>s.domain!==meta.domain);
    VHosts.remove(meta.domain, { persist: false });
  } else if(meta.type==='offline-prune'){
    state.offline.trees = state.offline.trees.filter(x=>!(x.svc===meta.svc && x.ver===meta.ver));
  } else if(meta.type==='restore'){
    VHosts.rebuildAll({ persist: false });
  }
  setTimeout(()=>{ render(); toast(t('toast.uiUpdated'),'ok',1400); },320);
}

function startEdit(el){
  if(el.classList.contains('editing')) return;
  const kind = el.dataset.kind, version = el.dataset.version || '', field = el.dataset.field, type = el.dataset.inline || 'text';
  const original = el.dataset.original || el.textContent.trim();
  let realValue = original;
  if(type === 'password'){ const key = `${kind.toUpperCase()}_${version.replace(/\./g,'')}_ROOT_PASSWORD`; realValue = state.env[key] || ''; }
  const inputCls = type === 'port' ? 'port-input' : (type === 'password' ? 'password-input' : 'text-input');
  const inputVal = type === 'password' ? realValue : (realValue === '—' ? '' : realValue);
  el.classList.add('editing');
  el.innerHTML = `<input type="text" class="${inputCls}" value="${esc(inputVal)}" spellcheck="false" autocomplete="off">`;
  const input = el.querySelector('input'); input.focus(); input.select();
  let committed = false;
  function cancel(){ if(committed) return; committed = true; el.classList.remove('editing'); el.innerHTML = originalHtml(el); setupInlineEdit(el.parentElement || document); }
  function commit(){
    if(committed) return;
    const newValue = input.value.trim();
    if(type === 'port' && !/^\d{1,5}$/.test(newValue)){ toast(t('common.invalidPort'), 'err'); input.focus(); return; }
    /* ★ FIX #2：移除密码长度校验 */
    if(newValue === realValue){ cancel(); return; }

    const check = preflight('update-config', { kind, version, field, newValue });
    if(!check.ok){ toast(check.errors.join('\n'), 'err', 4600); input.focus(); return; }

    committed = true; el.classList.remove('editing'); el.classList.add('saving');
    if(type === 'password'){ el.innerHTML = `<span class="pw">••••••••</span>`; el.dataset.original = '••••••••'; }
    else { el.innerHTML = `${esc(newValue)}`; el.dataset.original = newValue; }
    const meta = { type: 'update-config', kind, version, field, oldValue: realValue, newValue: newValue };
    runTask([kind, field, 'set', version, newValue], `${kind} ${version} · ${field} → ${newValue.length > 12 ? newValue.slice(0,8)+'…' : (newValue || '∅')}`, meta);
    setTimeout(()=>{ el.classList.remove('saving'); }, 4000);
  }
  input.addEventListener('keydown', (e)=>{ if(e.key === 'Enter'){ e.preventDefault(); commit(); } else if(e.key === 'Escape'){ e.preventDefault(); cancel(); } e.stopPropagation(); });
  input.addEventListener('blur', ()=>{ setTimeout(()=>{ if(!committed) cancel(); }, 100); });
}
function originalHtml(el){ const type = el.dataset.inline || 'text'; if(type === 'password') return `<span class="pw">••••••••</span>`; return esc(el.dataset.original || ''); }

function startPortEdit(el){
  if(el.classList.contains('editing')) return;
  const domain = el.dataset.domain, original = el.dataset.original || el.textContent.trim();
  el.classList.add('editing');
  el.innerHTML = `<input type="text" inputmode="numeric" value="${esc(original)}" spellcheck="false" autocomplete="off">`;
  const input = el.querySelector('input'); input.focus(); input.select();
  let committed = false;
  function cancel(){ if(committed) return; committed = true; el.classList.remove('editing'); el.innerHTML = esc(original); }
  function commit(){
    if(committed) return;
    const newValue = input.value.trim();
    if(!/^\d{1,5}$/.test(newValue)){ toast(t('common.invalidPort'), 'err'); input.focus(); return; }
    const newPort = parseInt(newValue, 10), oldPort = parseInt(original, 10);
    if(newPort === oldPort){ cancel(); return; }

    const check = preflight('site-port', { domain, newValue: newPort });
    if(!check.ok){ toast(check.errors.join('\n'), 'err', 4600); input.focus(); return; }

    const finalPort = (check.adjusted && check.adjusted.port != null) ? check.adjusted.port : newPort;
    if(check.warnings.length) toast(check.warnings.join('\n'), 'info', 3600);

    committed = true; el.classList.remove('editing'); el.classList.add('saving');
    el.innerHTML = esc(String(finalPort)); el.dataset.original = String(finalPort);
    runTask(['site', 'port', 'set', domain, String(finalPort)], `${domain} · 端口 ${oldPort} → ${finalPort}`, { type: 'site-port', domain, oldValue: oldPort, newValue: finalPort });
    setTimeout(()=>{ el.classList.remove('saving'); }, 4000);
  }
  input.addEventListener('keydown', (e)=>{ if(e.key === 'Enter'){ e.preventDefault(); commit(); } else if(e.key === 'Escape'){ e.preventDefault(); cancel(); } e.stopPropagation(); });
  input.addEventListener('blur', ()=>{ setTimeout(()=>{ if(!committed) cancel(); }, 100); });
}

function openRewriteModal(domain){
  const site = state.sites.find(s=>s.domain===domain);
  if(!site) return;

  const initialPreset = site.rewrite || 'none';
  const initialRule = (typeof site.rewriteRule === 'string' && site.rewriteRule.trim())
    ? site.rewriteRule
    : (REWRITE_PRESETS[initialPreset]?.rule || REWRITE_PRESETS.custom.rule);

  let selected = initialPreset;
  let currentRule = initialRule;
  let previewOpen = false;

  function buildFwGrid(){ return Object.entries(REWRITE_PRESETS).map(([key,p])=>`<div class="fw-card ${key===selected?'selected':''}" data-fw="${key}"><div class="fw-icon">${p.icon}</div><div class="fw-name">${esc(t(p.nameKey))}</div><div class="fw-tag">${esc(p.tag)}</div></div>`).join(''); }
  function refresh(){ $('#fw-grid', root).innerHTML = buildFwGrid(); $$('.fw-card', root).forEach(el=>{ el.addEventListener('click', ()=>{ selected=el.dataset.fw; currentRule=REWRITE_PRESETS[selected].rule; $('#rw-editor', root).value=currentRule; refresh(); }); }); updateDirtyState(); }
  function updateDirtyState(){
    const btn=$('#rw-apply',root);
    const editorVal=$('#rw-editor',root).value;
    const dirty = selected !== initialPreset || editorVal !== initialRule;
    btn.disabled=!dirty;
    btn.textContent = dirty ? t('rw.apply') : t('rw.current');
  }
  const root = openModalShell(`<div class="modal modal-lg" role="dialog" aria-modal="true"><button class="modal-close-x" data-modal-close type="button" aria-label="${esc(t('common.cancel'))}"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round"><path d="M18 6 6 18M6 6l12 12"/></svg></button><div class="modal-head"><h3>${esc(domain)} · ${t('rw.title')}</h3><p>${t('rw.subtitle')}</p></div><div class="modal-body"><div class="field"><label>${t('rw.select')}</label><div class="fw-grid" id="fw-grid"></div></div><div class="field"><label>${t('rw.rule')} <span style="font-weight:400;color:var(--text-mute);font-size:11.5px">${t('rw.rule.hint')}</span></label><textarea class="rw-editor" id="rw-editor" spellcheck="false"></textarea><div class="hint">${t('rw.writeHint', {domain: esc(domain), root: esc(sitesRoot())})}</div></div><div class="field"><button class="rw-collapse-btn collapsed" id="rw-preview-toggle"><span class="arrow">▼</span><span id="rw-preview-label">${t('rw.showPreview')}</span></button><div class="cmd-preview" id="rw-preview" style="display:none;white-space:pre-wrap;max-height:240px;overflow-y:auto"></div></div><div class="alert alert-info">${t('rw.applyHint')}</div></div><div class="modal-foot"><button class="btn" data-modal-close>${t('common.cancel')}</button><button class="btn btn-primary" id="rw-apply">${t('rw.apply')}</button></div><div class="modal-resizer"></div></div>`);
  $('#rw-editor', root).value = currentRule;
  refresh(); updateDirtyState();
  function escapeHtml(s){ return String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
  function renderPreview(){ const txt = $('#rw-editor', root).value; $('#rw-preview', root).innerHTML = escapeHtml(txt).replace(/(location|try_files|rewrite|if|return|last|break)\b/g,'<span style="color:var(--accent)">$1</span>').replace(/(\$[a-zA-Z_]+)/g,'<span style="color:var(--ok)">$1</span>').replace(/(#[^\n]*)/g,'<span style="color:var(--text-mute)">$1</span>'); }
  $('#rw-preview-toggle', root).addEventListener('click', ()=>{ previewOpen=!previewOpen; const btn=$('#rw-preview-toggle',root); btn.classList.toggle('collapsed',!previewOpen); $('#rw-preview-label',root).textContent=previewOpen?t('rw.hidePreview'):t('rw.showPreview'); const prev=$('#rw-preview',root); prev.style.display=previewOpen?'':'none'; if(previewOpen) renderPreview(); });
  $('#rw-editor', root).addEventListener('input', e=>{ currentRule=e.target.value; updateDirtyState(); if(previewOpen) renderPreview(); });
  $('#rw-editor', root).addEventListener('keydown', e=>{ if(e.key==='Tab'){ e.preventDefault(); const ta=e.target; const start=ta.selectionStart, end=ta.selectionEnd; ta.value=ta.value.slice(0,start)+'    '+ta.value.slice(end); ta.selectionStart=ta.selectionEnd=start+4; currentRule=ta.value; updateDirtyState(); } });
  $$('[data-modal-close]', root).forEach(b=>b.addEventListener('click', closeModal));
  $('#rw-apply', root).addEventListener('click', ()=>{
    const finalRule=$('#rw-editor',root).value;
    const preset=REWRITE_PRESETS[selected];

    const check = preflight('rewrite', { domain, rule: finalRule });
    if(!check.ok){ toast(check.errors.join('\n'), 'err', 4600); return; }

    closeModal();
    const ruleToStore = (preset && finalRule === preset.rule) ? '' : finalRule;
    runTask(['site','rewrite',domain,'--preset',selected],`${domain} · ${t(preset.nameKey)}`,{ type:'rewrite', domain, preset:selected, rule: ruleToStore });
  });
}

function openConfigModal(kind, version){
  const meta = SVC_META[kind];
  const files = getDefaultFiles(kind, version);
  if(!files.length){ toast('No config files', 'info'); return; }
  let activeFile = files[0].name;
  const originals = {}; const drafts = {};
  files.forEach(f => { const c = getFileContent(kind, version, f.name); originals[f.name] = c; drafts[f.name] = c; });
  function isModified(name){ return drafts[name] !== originals[name]; }
  function modifiedCount(){ return files.filter(f => isModified(f.name)).length; }
  function currentFile(){ return files.find(f => f.name === activeFile); }
  function fileListHtml(){ return files.map(f => { const mod = isModified(f.name) ? ' modified' : ''; const active = f.name === activeFile ? ' active' : ''; return `<button class="config-file${active}${mod}" data-file="${esc(f.name)}"><svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" style="flex-shrink:0"><path d="M14 3v5h5"/><path d="M14 3H6a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2V8z"/></svg><span class="file-name">${esc(f.name)}</span><span class="file-dot"></span></button>`; }).join(''); }
  function editorHtml(){ const f = currentFile(); const full = `${state.env.PHPO_HOME}/${f.path}`; return `<div class="config-editor-head"><span class="path-text" title="${esc(full)}">${esc(full)}</span><span style="font-size:11.5px;color:var(--text-mute);flex-shrink:0">${esc(t('config.editorHint'))}</span></div><textarea class="config-editor" id="config-editor" spellcheck="false"></textarea>`; }
  const root = openModalShell(`<div class="modal modal-xl" role="dialog" aria-modal="true"><button class="modal-close-x" data-modal-close type="button" aria-label="${esc(t('common.cancel'))}"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round"><path d="M18 6 6 18M6 6l12 12"/></svg></button><div class="modal-head"><h3>${esc(t(meta.titleKey))} ${esc(version)} · ${t('config.title')}</h3><p>${t('config.subtitle')}</p></div><div class="modal-body config-body"><div class="config-layout"><aside class="config-files" id="config-files"><div class="config-files-label">${t('config.files')}</div>${fileListHtml()}</aside><div class="config-editor-wrap" id="config-editor-wrap">${editorHtml()}</div></div><div style="padding:12px 22px;border-top:1px solid var(--border-2);background:var(--surface-2)"><div class="alert alert-info" style="padding:8px 12px">${t('config.hint')}</div><div style="margin-top:8px">${mountListHtml(kind, version)}</div></div></div><div class="modal-foot"><span style="flex:1;font-size:12px;color:var(--text-mute);font-family:var(--mono)" id="config-status"></span><button class="btn" id="config-reset">${t('config.reset')}</button><button class="btn" data-modal-close>${t('common.cancel')}</button><button class="btn btn-primary" id="config-apply">${t('config.apply')}</button></div><div class="modal-resizer"></div></div>`);
  const editorEl = () => $('#config-editor', root); const statusEl = () => $('#config-status', root); const applyBtn = () => $('#config-apply', root); const resetBtn = () => $('#config-reset', root);
  function syncStatus(){ const n = modifiedCount(); if(n > 0){ statusEl().innerHTML = `<span style="color:var(--warn)">● ${t('config.unsaved')}</span> · ${n}/${files.length}`; applyBtn().disabled = false; applyBtn().textContent = t('config.applyCount', {count: n}); resetBtn().disabled = false; } else { statusEl().textContent = ''; applyBtn().disabled = true; applyBtn().textContent = t('config.apply'); resetBtn().disabled = true; } }
  function loadEditor(){ const ta = editorEl(); ta.value = drafts[activeFile] || ''; ta.scrollTop = 0; syncStatus(); }
  function bindFiles(){ $$('.config-file', root).forEach(btn => { btn.addEventListener('click', () => { drafts[activeFile] = editorEl().value; activeFile = btn.dataset.file; $('#config-files', root).innerHTML = `<div class="config-files-label">${t('config.files')}</div>` + fileListHtml(); bindFiles(); loadEditor(); }); }); }
  loadEditor(); bindFiles();
  root.addEventListener('input', e => { if(e.target && e.target.id === 'config-editor'){ drafts[activeFile] = e.target.value; syncStatus(); $$('.config-file', root).forEach(btn => { const nm = btn.dataset.file; btn.classList.toggle('modified', isModified(nm)); }); } });
  root.addEventListener('keydown', e => { if(e.target && e.target.id === 'config-editor' && e.key === 'Tab'){ e.preventDefault(); const ta = e.target; const start = ta.selectionStart, end = ta.selectionEnd; ta.value = ta.value.slice(0, start) + '    ' + ta.value.slice(end); ta.selectionStart = ta.selectionEnd = start + 4; drafts[activeFile] = ta.value; syncStatus(); } }, true);
  resetBtn().addEventListener('click', () => { drafts[activeFile] = originals[activeFile]; loadEditor(); $$('.config-file', root).forEach(btn => { btn.classList.toggle('modified', isModified(btn.dataset.file)); }); });
  applyBtn().addEventListener('click', () => {
    drafts[activeFile] = editorEl().value;
    const changedFiles = files.filter(f => isModified(f.name));
    if(!changedFiles.length){ toast(t('config.noChanges'), 'info', 1400); return; }

    const check = preflight('service-config', { kind, version, files: changedFiles.map(f=>f.name) });
    if(!check.ok){ toast(check.errors.join('\n'), 'err', 4600); return; }

    changedFiles.forEach(f => { state.configs[`${kind}:${version}:${f.name}`] = drafts[f.name]; });
    persistState();
    const fileNames = changedFiles.map(f => f.name);
    closeModal();
    runTask([kind, 'config', 'save', version, '--files', fileNames.join(',')], `${kind} ${version} · ${t('config.title')} (${fileNames.length})`, { type:'service-config', kind, version, files: fileNames });
    toast(t('config.saved', {kind, version, count: fileNames.length}), 'ok', 2600);
  });
  $$('[data-modal-close]', root).forEach(b => b.addEventListener('click', closeModal));
}

    });
  }

    if(zoomVal) zoomVal.textContent = Math.round(layout.scale * 100) + '%';
  }); }

  $$('#lang-picker .pick', document).forEach(btn=>{ btn.addEventListener('click', ()=>{ i18n.setLocale(btn.dataset.lang); render(); updateTray(); toast(t('toast.langSwitched', {lang: btn.dataset.lang === 'zh-CN' ? '简体中文' : 'English'}), 'ok', 1600); }); });
  const trayMin = $('#tray-minimize'); if(trayMin) trayMin.addEventListener('change', ()=>{ state.tray.minimizeOnClose = trayMin.checked; persistState(); });
  const trayShow = $('#tray-show'); if(trayShow) trayShow.addEventListener('change', ()=>{ state.tray.enabled = trayShow.checked; $('#tray-sim').style.display = trayShow.checked ? '' : 'none'; persistState(); });
  markActivePreset();
  markActiveZoom();
}
function downloadBackup(file){
  const content = `# phpo backup archive (demo)\n# file: ${file}\n# restore: phpo restore ${state.env.BACKUP_ROOT}/${file}\n`;
  const blob = new Blob([content], {type:'application/gzip'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a'); a.href = url; a.download = file;
  document.body.appendChild(a); a.click(); a.remove();
  URL.revokeObjectURL(url);
  toast(t('toast.backupDownloaded',{file}), 'ok', 3600);
}

function updateTray(){
  const theme = THEMES.find(th => th.id === state.theme) || THEMES[0];
  const themeName = t(theme.nameKey);
  const labelEl = $('#tray-label');
  if(labelEl) labelEl.textContent = `phpo · ${themeName}`;
  const dot = $('#tray-dot');
  if(dot){
    dot.style.background = theme.colors[2];
    if(state.task?.status === TASK_STATUS.RUNNING) dot.className = 'tray-dot running';
    else if(state.task?.status === TASK_STATUS.FAILED) dot.className = 'tray-dot error';
    else dot.className = 'tray-dot';
  }
}

  const p = $('#cmd-palette'); p.classList.add('open');
  const input = $('#cmd-input'); input.value = ''; input.placeholder = t('common.cmdPalette') + '…';

function renderCmdList(q){
  const query = q.trim().toLowerCase();
  const items = CMD_ITEMS.filter(it=>{ if(!query) return true; const label = t(it.labelKey).toLowerCase(); return label.includes(query) || it.labelKey.toLowerCase().includes(query); });
  const list = $('#cmd-list');
  list.innerHTML = items.map(it=>{ const idx = CMD_ITEMS.indexOf(it); return `<div class="cmd-palette-item" data-idx="${idx}"><span>${esc(t(it.labelKey))}</span>${it.kbd ? `<span class="kbd">${it.kbd}</span>` : ''}</div>`; }).join('');

}

function bindEvents(){
  $('#nav').addEventListener('click', e=>{ const btn = e.target.closest('[data-route]'); if(!btn) return; state.route = btn.dataset.route; render(); $('#view').scrollTop = 0; });
  $('#view').addEventListener('click', e=>{
    if(e.target.closest('.site-domain')) return;
    if(e.target.closest('.inline-edit')) return;
    if(e.target.closest('.port-edit')) return;
    const jump = e.target.closest('[data-route-jump]'); if(jump){ state.route = jump.dataset.routeJump; render(); $('#view').scrollTop = 0; return; }
    const btn = e.target.closest('[data-action]'); if(!btn) return;
    const d = btn.dataset;
    switch(d.action){
      case 'install':         openInstallModal(d.kind); break;
      case 'uninstall':       openUninstallModal(d.kind, d.version); break;
      case 'stop-service': {
        const check = preflight('service-stop', { kind:d.kind, version:d.version });
        if(!check.ok){ toast(check.errors.join('\n'), 'err', 4600); break; }
        if(check.warnings.length){
          openDangerConfirm({
            title: `${t('svc.stop')} ${d.kind} ${d.version}`,
            warnings: check.warnings.map(w=>({ text:w })),
            confirmLabel: t('common.confirm'),
            onConfirm: ()=> runTask([d.kind,'stop',d.version], `${t('svc.stop')} ${d.kind} ${d.version}`, { type:'service-stop', kind:d.kind, version:d.version }),
          });
        } else {
          runTask([d.kind,'stop',d.version], `${t('svc.stop')} ${d.kind} ${d.version}`, { type:'service-stop', kind:d.kind, version:d.version });
        }
        break;
      }
      case 'start-service': {
        const check = preflight('service-start', { kind:d.kind, version:d.version });
        if(!check.ok){ toast(check.errors.join('\n'), 'err', 4600); break; }
        runTask([d.kind,'start',d.version], `${t('svc.start')} ${d.kind} ${d.version}`, { type:'service-start', kind:d.kind, version:d.version });
        break;
      }
      case 'site-add':        openSiteAddModal(); break;
      case 'site-remove':     openSiteRemoveModal(d.domain); break;
      case 'restore':         openRestoreModal(d.file); break;
      case 'delete-backup':   openDeleteBackupModal(d.file); break;
      case 'download-backup': downloadBackup(d.file); break;
      case 'offline-prune':   openOfflinePruneModal(d.svc, d.ver, d.size); break;
      case 'php-extensions':  openPhpExtensionsModal(d.version); break;
      case 'service-config':  openConfigModal(d.kind, d.version); break;
      case 'row-menu':        e.stopPropagation(); openRowMenu(btn, d.domain); break;
      case 'hosts-add':       toast(t('sites.hosts.toast',{domain:d.domain}), 'ok'); state.sites.find(s=>s.domain===d.domain).hosts = true; persistState(); render(); break;
      case 'open-path':       copyText(d.path).then(ok=>toast(t('sites.path.toast',{path:d.path}), ok?'ok':'info', 3200)); break;
      case 'run-task': {
        const args = d.args.split(',').filter(Boolean);
        const pfAction = d.pfAction;
        if(pfAction){
          runGuardedTask(pfAction, {}, args, d.label || args.join(' '));
        } else {
          runTask(args, d.label || args.join(' '));
        }
        break;
      }
    }
  });
  $('#view').addEventListener('change', e=>{
    const sel = e.target.closest('.php-select'); if(!sel) return;
    const domain = sel.dataset.domain, newV = sel.value;
    const site = state.sites.find(s=>s.domain===domain); if(!site || site.php === newV) return;

    const check = preflight('php-switch', { domain, newPhp: newV });
    if(!check.ok){
      toast(check.errors.join('\n'), 'err', 3400);
      sel.value = site.php;
      return;
    }

    const oldPhp = site.php;
    sel.disabled = true;
    if(sel.selectedIndex >= 0) sel.options[sel.selectedIndex].textContent = '…';
    runTask(['site','php','set',domain,newV], `${domain} · PHP ${oldPhp} → ${newV}`, { type:'php-switch', domain, oldPhp, newPhp: newV });
  });
  $('#view').addEventListener('scroll', closeRowMenu, {passive:true});
  window.addEventListener('resize', closeRowMenu);
  $('#refresh-btn').addEventListener('click', ()=>toast(t('common.refresh'),'ok',1400));
  $('#theme-btn').addEventListener('click', openThemePicker);
  $('#drawer-toggle').addEventListener('click', toggleDrawer);
  $('#drawer-cancel').addEventListener('click', cancelTask);
  $('#drawer-copy').addEventListener('click', ()=>{ if(!state.task){ toast(t('task.noLog'),'info',1200); return; } const txt = state.task.lines.slice(0, state.task.cursor).map(l=>l.s).join('\n'); copyText(txt).then(ok=>toast(ok?t('task.copyLog'):t('common.copyFailed'), ok?'ok':'err', 1600)); });
  $('#modal-root').addEventListener('click', e=>{ if(e.target.id === 'modal-root') closeModal(); if(e.target.closest('[data-modal-close]')) closeModal(); });
  $('#cmd-palette').addEventListener('click', e=>{ if(e.target.id === 'cmd-palette') closeCmdPalette(); });
  $('#tray-sim').addEventListener('click', openTrayMenu);
  document.addEventListener('click', e=>{ if(!e.target.closest('#tray-sim') && !e.target.closest('#tray-menu')){ $('#tray-menu').classList.remove('open'); } if(!e.target.closest('#row-menu') && !e.target.closest('[data-action="row-menu"]')){ closeRowMenu(); } });
  document.addEventListener('keydown', e=>{
    const modalOpen = $('#modal-root').classList.contains('open');
    const paletteOpen = $('#cmd-palette').classList.contains('open');
    if(e.key === 'Escape'){ if(paletteOpen){ closeCmdPalette(); return; } if(modalOpen){ closeModal(); return; } closeRowMenu(); return; }
    const mod = e.ctrlKey || e.metaKey;
    if(mod && e.key.toLowerCase() === 'k'){ e.preventDefault(); paletteOpen ? closeCmdPalette() : openCmdPalette(); return; }
    if(mod && e.key.toLowerCase() === 'r'){ e.preventDefault(); toast(t('common.refresh'),'ok',1200); return; }
    if(mod && (e.key === '=' || e.key === '+')){ e.preventDefault(); layout.zoomIn(); return; }
    if(mod && (e.key === '-' || e.key === '_')){ e.preventDefault(); layout.zoomOut(); return; }
    if(mod && e.key === '0'){ e.preventDefault(); layout.zoomReset(); return; }
    if(mod && /^[1-9]$/.test(e.key)){ const idx = parseInt(e.key)-1; const routes = ['sites','php','mysql','pgsql','redis','nginx','backup','overview','settings']; if(routes[idx]){ e.preventDefault(); state.route = routes[idx]; render(); $('#view').scrollTop = 0; } }
  });
  window.addEventListener('beforeunload', e=>{
    if(state.task && state.task.status === TASK_STATUS.RUNNING){
      e.preventDefault();
      e.returnValue = '';
    }
  });
}
